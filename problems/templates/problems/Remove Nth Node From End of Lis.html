Remove Nth Node From End of List

Given the head of a linked list, remove the nth node from the end of the list and return its head. We use a dummy node and a two-pointer approach to achieve this in one pass.

Solution

def removeNthFromEnd(head: Optional[ListNode], n: int) -> Optional[ListNode]:
    # Dummy node handles edge cases like deleting the head
    dummy = ListNode(0, head)
    left = dummy
    right = head
    
    # Step 1: Move right pointer n steps ahead
    for _ in range(n):
        right = right.next
        
    # Step 2: Move both pointers until right reaches the end
    while right:
        left = left.next
        right = right.next
        
    # Step 3: Delete the target node
    left.next = left.next.next
    
    return dummy.next

ListNode* removeNthFromEnd(ListNode* head, int n) {
    ListNode* dummy = new ListNode(0, head);
    ListNode* left = dummy;
    ListNode* right = head;
    
    // Step 1: Move right pointer n steps ahead
    for (int i = 0; i < n; i++) {
        right = right->next;
    }
    
    // Step 2: Move both pointers
    while (right != nullptr) {
        left = left->next;
        right = right->next;
    }
    
    // Step 3: Delete the target node
    ListNode* toDelete = left->next;
    left->next = left->next->next;
    delete toDelete;
    
    ListNode* newHead = dummy->next;
    delete dummy;
    return newHead;
}

public ListNode removeNthFromEnd(ListNode head, int n) {
    ListNode dummy = new ListNode(0, head);
    ListNode left = dummy;
    ListNode right = head;
    
    // Step 1: Move right pointer n steps ahead
    for (int i = 0; i < n; i++) {
        right = right.next;
    }
    
    // Step 2: Move both pointers
    while (right != null) {
        left = left.next;
        right = right.next;
    }
    
    // Step 3: Delete the target node
    left.next = left.next.next;
    
    return dummy.next;
}
